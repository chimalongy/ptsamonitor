-- ─────────────────────────────────────────────────────────────────────────────
-- PIE CHART (Transaction Count & Amount by Response Code)
-- ─────────────────────────────────────────────────────────────────────────────
SELECT 
    COUNT(1)            AS TXN_COUNT,
    SUM(AMOUNT)         AS TOTAL_AMOUNT,
    RESPCODE,
    RESPCODE_DESCRIPTION
FROM PTSA.MV_DASHBOARD_CACHE
GROUP BY RESPCODE, RESPCODE_DESCRIPTION
ORDER BY TXN_COUNT DESC;


-- ─────────────────────────────────────────────────────────────────────────────
-- DONUT CHART (Failed Transactions by Status Description)
-- ─────────────────────────────────────────────────────────────────────────────
SELECT 
    RESPCODE_DESCRIPTION,
    COUNT(DISTINCT ID)  AS TXN_COUNT
FROM PTSA.MV_DASHBOARD_CACHE
WHERE RESPCODE NOT IN (0, 00)
GROUP BY RESPCODE_DESCRIPTION
ORDER BY RESPCODE_DESCRIPTION;


-- ─────────────────────────────────────────────────────────────────────────────
-- BAR CHART (Transactions by Institution)
-- ─────────────────────────────────────────────────────────────────────────────
SELECT 
    COUNT(1)            AS TXN_COUNT,
    INST
FROM PTSA.MV_DASHBOARD_CACHE
GROUP BY INST
ORDER BY TXN_COUNT DESC;

================================================================================
  NOTES
================================================================================

  - All queries target PTSA.MV_DASHBOARD_CACHE (no live DATAMART scans)
  - MV is refreshed every 5 minutes via PTSA.REFRESH_MV_DASHBOARD_CACHE job
  - MV covers only TODAY's transactions (TIME >= TRUNC(SYSDATE) AND TIME < TRUNC(SYSDATE)+1)
  - All filter columns are indexed: RESPCODE, INSTITUTION_CODE, SOURCE1, TIME
  - RESPCODE = 0 means approved; all others are failures (used in DONUT chart)
  - MASKEDPAN is used in the transaction table instead of raw PAN for security
