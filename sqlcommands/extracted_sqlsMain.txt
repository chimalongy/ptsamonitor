================================================================================
UNIFIED PAYMENTS
================================================================================

-- Pie Chart
SELECT COUNT(1) AS count, SUM(amount) AS amount, respcode
FROM PTSA.DATAMART
WHERE time >= '<startDate>'
  AND time < '<stopDate>'
  -- AND SOURCE='PTSA'   /* UPTSA 1 filter */
  -- AND SOURCE='TWO'    /* UPTSA 2 filter */
GROUP BY respcode;

--------------------------------------------------------------------------------

-- Donut Chart (Status Code)
SELECT NVL(r.DECRIPTION, 'Error') AS DECRIPTION,
       COUNT(DISTINCT p.id) AS txn_count
FROM PTSA.DATAMART p
LEFT JOIN PTSA.APPROVAL2 r ON p.respcode = r.code
WHERE p.time > '<startDate>'
  AND p.time < '<stopDate>'
  AND p.respcode NOT IN ('00', '0')
  -- AND SOURCE='PTSA'   /* UPTSA 1 filter */
  -- AND SOURCE='TWO'    /* UPTSA 2 filter */
GROUP BY NVL(r.DECRIPTION, 'Error')
ORDER BY DECRIPTION;

--------------------------------------------------------------------------------

-- Bar Chart (Transactions by Institution)
SELECT 
    COUNT(1) AS count,
    NVL(DESTINATION, AUTHFINAME) AS inst
FROM PTSA.DATAMART
WHERE time > '<startDate>'
  -- AND SOURCE='PTSA'   /* UPTSA 1 filter */
  -- AND SOURCE='TWO'    /* UPTSA 2 filter */
GROUP BY NVL(DESTINATION, AUTHFINAME);

--------------------------------------------------------------------------------

-- Transaction Table (Main Query)
SELECT SOURCE,
       TRANNUMBER AS RRN,
       PAN,
       TERMNAME AS TERMINALID,
       RESPCODE,
       TIME,
       TERMRETAILERNAME AS MERCHANTID,
       DESTINATION
FROM PTSA.DATAMART
WHERE TIME >= TO_DATE('<startDate>', 'YYYY-MM-DD')
  AND TIME < TO_DATE('<endDate>', 'YYYY-MM-DD')
  -- AND PAN LIKE '%<search>%' OR RRN LIKE '%<search>%'   /* search filter */
  -- AND SOURCE = '<source>'                               /* source filter */
  -- AND TERMNAME = '<terminalId>'                         /* terminal filter */
  -- AND TERMRETAILERNAME = '<merchantId>'                 /* merchant filter */
ORDER BY TIME DESC
OFFSET <offset> ROWS FETCH NEXT <limit> ROWS ONLY;

-- Transaction Table (Count Query)
SELECT COUNT(*) AS TOTAL
FROM PTSA.DATAMART
WHERE TIME >= TO_DATE('<startDate>', 'YYYY-MM-DD')
  AND TIME < TO_DATE('<endDate>', 'YYYY-MM-DD');

================================================================================
DATAMART COUNTS
================================================================================

-- PTSA Count
SELECT COUNT(1) AS count
FROM PTSA.DATAMART
WHERE time >= TO_DATE('<startDate>', 'DD-MON-YYYY')
  AND time < TO_DATE('<stopDate>', 'DD-MON-YYYY')
  AND source = 'PTSA';

--------------------------------------------------------------------------------

-- TWO Count
SELECT COUNT(1) AS count
FROM PTSA.DATAMART
WHERE time >= TO_DATE('<startDate>', 'DD-MON-YYYY')
  AND time < TO_DATE('<stopDate>', 'DD-MON-YYYY')
  AND source = 'TWO';

--------------------------------------------------------------------------------

-- CBN Update Count
SELECT COUNT(1) AS count
FROM PTSA.DATAMART
WHERE time >= TO_DATE('<startDate>', 'DD-MON-YYYY')
  AND time < TO_DATE('<stopDate>', 'DD-MON-YYYY')
  AND cbnupdate = 'Y';
