# 1. SSH into your app server
ssh user@your-app-server-ip

# 2. Install Redis
sudo apt update
sudo apt install redis-server -y

# 3. Configure Redis
sudo nano /etc/redis/redis.conf

# Bind to localhost only — C# app is on same server
bind 127.0.0.1

# Set a password
requirepass YourStrongPasswordHere

# Limit memory
maxmemory 2gb
maxmemory-policy allkeys-lru

# Disable persistence — it's a cache not a database
save ""
appendonly no

# 4. Restart and enable Redis
sudo systemctl restart redis-server
sudo systemctl enable redis-server   # auto-start on server reboot

# 5. Verify
redis-cli ping
# PONG

redis-cli -a YourStrongPasswordHere ping
# PONG
